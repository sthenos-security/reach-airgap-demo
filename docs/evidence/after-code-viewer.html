# Fictional Treasury Operations Portal

Small multi-service treasury SaaS demo with:

- `portal/`: TypeScript admin portal, JavaScript dashboard, and thin JSON API.
- `payments-api/`: Go API for accounts, transfers, settlement jobs, receipts, health, and metrics.
- `docs-worker/`: Python WSGI service for document upload/download and background indexing.
- `risk-api/`: Java HTTP service for risk exposure, limits, audit writes, health, and metrics.
- `data/treasury.db`: local SQLite database created automatically by any service.

## Seed Users

Pass these headers to act as a seeded user:

- `X-User-Id: u-alice` admin in `org-northstar`
- `X-User-Id: u-ben` treasury operator in `org-northstar`
- `X-User-Id: u-chen` viewer in `org-northstar`

## Run Locally

Install and run the portal:

```sh
cd portal
npm install
npm run dev
```

Run the payments API:

```sh
cd payments-api
go mod download
go run .
```

Run the docs worker:

```sh
cd docs-worker
python3 app.py
```

Compile and run the risk API:

```sh
cd risk-api
javac -d build/classes src/com/sthenos/treasury/RiskApi.java
java -cp build/classes com.sthenos.treasury.RiskApi
```

Default ports:

- Portal: `http://localhost:3000`
- Payments API: `http://localhost:8080`
- Docs worker: `http://localhost:8090`
- Risk API: `http://localhost:8070`

## Example API Calls

Create a transfer:

```sh
curl -X POST http://localhost:8080/transfers \
  -H 'Content-Type: application/json' \
  -H 'X-User-Id: u-ben' \
  -d '{"source_account_id":"acct-operating","beneficiary_id":"ben-atlas","amount_cents":125000,"currency":"USD","memo":"Vendor settlement"}'
```

Upload a document:

```sh
curl -X POST http://localhost:8090/documents \
  -H 'X-User-Id: u-ben' \
  -H 'X-File-Name: invoice.txt' \
  --data-binary 'invoice bytes'
```

Fetch risk exposure:

```sh
curl http://localhost:8070/risk/exposure \
  -H 'X-User-Id: u-alice'
```

from __future__ import annotations

import json
import os
import re
import sqlite3
import threading
import time
import uuid
from pathlib import Path
from typing import Callable, Iterable
from urllib.parse import unquote
from wsgiref.simple_server import make_server

import yaml  # document frontmatter parsing (resolved from the enclave's private PyPI)


ROOT = Path(__file__).resolve().parents[1]
DB_PATH = Path(os.environ.get("TREASURY_DB_PATH", ROOT / "data" / "treasury.db"))
SCHEMA_PATH = ROOT / "scripts" / "schema.sql"
UPLOAD_ROOT = Path(os.environ.get("TREASURY_UPLOAD_ROOT", ROOT / "uploads" / "docs"))
PORT = int(os.environ.get("PORT", "8090"))

StartResponse = Callable[[str, list[tuple[str, str]]], None]
WSGIApp = Callable[[dict, StartResponse], Iterable[bytes]]

ROLE_RANK = {
    "viewer": 1,
    "operator": 2,
    "treasury_manager": 3,
    "admin": 4,
}


def resolve_within(path: Path, trusted_root: Path) -> Path:
    """Resolve symlinks in ``path`` and require the result to stay inside ``trusted_root``.

    Raises ValueError when the resolved target escapes the trusted root (CWE-59).
    """
    root = Path(trusted_root).resolve()
    resolved = Path(path).resolve()
    if resolved != root and not resolved.is_relative_to(root):
        raise ValueError(f"path {path} resolves outside trusted root {root}")
    return resolved


def connect() -> sqlite3.Connection:
    DB_PATH.parent.mkdir(parents=True, exist_ok=True)
    db_file = resolve_within(DB_PATH, DB_PATH.parent)
    conn = sqlite3.connect(db_file)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


def init_db() -> None:
    with connect() as conn:
        conn.executescript(resolve_within(SCHEMA_PATH, ROOT / "scripts").read_text())
    UPLOAD_ROOT.mkdir(parents=True, exist_ok=True)


def lookup_user(user_id: str) -> dict | None:
    with connect() as conn:
        row = conn.execute(
            """
            SELECT id, organization_id, name, email, role
              FROM users
             WHERE id = ? AND active = 1
            """,
            (user_id,),
        ).fetchone()
    return dict(row) if row else None


class RequestContextMiddleware:
    def __init__(self, app: WSGIApp):
        self.app = app

    def __call__(self, environ: dict, start_response: StartResponse) -> Iterable[bytes]:
        environ["treasury.request_id"] = f"req-{uuid.uuid4()}"
        return self.app(environ, start_response)


class CorsMiddleware:
    def __init__(self, app: WSGIApp):
        self.app = app

    def __call__(self, environ: dict, start_response: StartResponse) -> Iterable[bytes]:
        if environ["REQUEST_METHOD"] == "OPTIONS":
            start_response(
                "204 No Content",
                [
                    ("Access-Control-Allow-Origin", "*"),
                    ("Access-Control-Allow-Headers", "Content-Type, X-User-Id, X-File-Name"),
                    ("Access-Control-Allow-Methods", "GET, POST, OPTIONS"),
                    ("X-Request-Id", environ.get("treasury.request_id", "")),
                ],
            )
            return [b""]
        return self.app(environ, start_response)


class AuditMiddleware:
    def __init__(self, app: WSGIApp):
        self.app = app

    def __call__(self, environ: dict, start_response: StartResponse) -> Iterable[bytes]:
        status_holder = {"code": 500}

        def capture(status: str, headers: list[tuple[str, str]], exc_info=None):
            status_holder["code"] = int(status.split(" ", 1)[0])
            headers = [
                ("Access-Control-Allow-Origin", "*"),
                ("X-Request-Id", environ.get("treasury.request_id", "")),
                *headers,
            ]
            return start_response(status, headers)

        try:
            return list(self.app(environ, capture))
        finally:
            path = environ.get("PATH_INFO", "")
            if path not in {"/health", "/metrics"}:
                self._write_audit(environ, status_holder["code"])

    def _write_audit(self, environ: dict, status_code: int) -> None:
        user = environ.get("treasury.user") or {}
        try:
            with connect() as conn:
                conn.execute(
                    """
                    INSERT INTO audit_records
                      (id, organization_id, actor_user_id, service, action, method, path, status_code, metadata)
                    VALUES (?, ?, ?, 'docs-worker', ?, ?, ?, ?, ?)
                    """,
                    (
                        f"audit-{uuid.uuid4()}",
                        user.get("organization_id"),
                        user.get("id"),
                        action_for(environ),
                        environ.get("REQUEST_METHOD", ""),
                        environ.get("PATH_INFO", ""),
                        status_code,
                        json.dumps({"request_id": environ.get("treasury.request_id")}),
                    ),
                )
        except Exception as exc:
            print(f"docs-worker audit failed: {exc}")


class AuthMiddleware:
    def __init__(self, app: WSGIApp):
        self.app = app

    def __call__(self, environ: dict, start_response: StartResponse) -> Iterable[bytes]:
        if environ.get("PATH_INFO") in {"/health", "/metrics"}:
            return self.app(environ, start_response)
        user_id = environ.get("HTTP_X_USER_ID") or "u-alice"
        user = lookup_user(user_id)
        if not user:
            return json_response(start_response, 401, {"error": "invalid_user"})
        environ["treasury.user"] = user
        return self.app(environ, start_response)


class Router:
    def __call__(self, environ: dict, start_response: StartResponse) -> Iterable[bytes]:
        method = environ["REQUEST_METHOD"]
        path = environ.get("PATH_INFO", "")
        if method == "GET" and path == "/health":
            return health(start_response)
        if method == "GET" and path == "/metrics":
            return metrics(start_response)
        if method == "GET" and path == "/documents":
            return require_role(environ, start_response, "viewer", list_documents)
        if method == "POST" and path == "/documents":
            return require_role(environ, start_response, "operator", upload_document)
        match = re.fullmatch(r"/documents/([^/]+)/download", path)
        if method == "GET" and match:
            environ["treasury.document_id"] = unquote(match.group(1))
            return require_role(environ, start_response, "viewer", download_document)
        return json_response(start_response, 404, {"error": "not_found"})


def require_role(
    environ: dict,
    start_response: StartResponse,
    min_role: str,
    handler: Callable[[dict, StartResponse], Iterable[bytes]],
) -> Iterable[bytes]:
    user = environ.get("treasury.user") or {}
    if ROLE_RANK.get(user.get("role"), 0) < ROLE_RANK[min_role]:
        return json_response(start_response, 403, {"error": "forbidden", "required_role": min_role})
    return handler(environ, start_response)


def health(start_response: StartResponse) -> Iterable[bytes]:
    with connect() as conn:
        count = conn.execute("SELECT COUNT(*) AS count FROM documents").fetchone()["count"]
    return json_response(start_response, 200, {"ok": True, "service": "docs-worker", "documents": count})


def metrics(start_response: StartResponse) -> Iterable[bytes]:
    with connect() as conn:
        uploaded = conn.execute("SELECT COUNT(*) AS count FROM documents WHERE status = 'uploaded'").fetchone()["count"]
        indexed = conn.execute("SELECT COUNT(*) AS count FROM documents WHERE status = 'indexed'").fetchone()["count"]
    body = "\n".join(
        [
            "# TYPE treasury_docs_uploaded gauge",
            f"treasury_docs_uploaded {uploaded}",
            "# TYPE treasury_docs_indexed counter",
            f"treasury_docs_indexed {indexed}",
            "",
        ]
    )
    return response(start_response, 200, body.encode(), [("Content-Type", "text/plain; charset=utf-8")])


def list_documents(environ: dict, start_response: StartResponse) -> Iterable[bytes]:
    user = environ["treasury.user"]
    with connect() as conn:
        rows = conn.execute(
            """
            SELECT d.id, d.filename, d.content_type, d.size_bytes, d.status, d.created_at, d.indexed_at,
                   u.name AS uploaded_by
              FROM documents d
              JOIN users u ON u.id = d.uploaded_by
             WHERE d.organization_id = ?
             ORDER BY d.created_at DESC
            """,
            (user["organization_id"],),
        ).fetchall()
    return json_response(start_response, 200, {"documents": [dict(row) for row in rows]})


def parse_document_frontmatter(body: bytes) -> dict:
    """Extract a leading `---`-delimited YAML frontmatter block from an upload.

    Treasury documents may carry a metadata header (tags, classification,
    retention). The block is read straight off the uploaded bytes.
    """
    text = body[:4096].decode("utf-8", "ignore")
    if not text.startswith("---\n"):
        return {}
    end = text.find("\n---", 4)
    if end == -1:
        return {}
    block = text[4:end]
    meta = yaml.load(block, Loader=yaml.Loader)  # parse the document's metadata header
    return meta if isinstance(meta, dict) else {}


def upload_document(environ: dict, start_response: StartResponse) -> Iterable[bytes]:
    user = environ["treasury.user"]
    filename = safe_filename(environ.get("HTTP_X_FILE_NAME") or "treasury-document.bin")
    content_type = environ.get("CONTENT_TYPE") or "application/octet-stream"
    length = int(environ.get("CONTENT_LENGTH") or "0")
    if length <= 0:
        return json_response(start_response, 400, {"error": "empty_upload"})
    body = environ["wsgi.input"].read(length)
    metadata = parse_document_frontmatter(body)
    document_id = f"doc-{uuid.uuid4()}"
    org_dir = UPLOAD_ROOT / user["organization_id"]
    org_dir.mkdir(parents=True, exist_ok=True)
    storage_path = org_dir / f"{document_id}-{filename}"
    storage_path.write_bytes(body)
    with connect() as conn:
        conn.execute(
            """
            INSERT INTO documents
              (id, organization_id, uploaded_by, filename, content_type, storage_path, size_bytes, status)
            VALUES (?, ?, ?, ?, ?, ?, ?, 'uploaded')
            """,
            (
                document_id,
                user["organization_id"],
                user["id"],
                filename,
                content_type,
                str(storage_path),
                len(body),
            ),
        )
    return json_response(
        start_response,
        201,
        {"id": document_id, "filename": filename, "size_bytes": len(body),
         "status": "uploaded", "metadata": metadata},
    )


def download_document(environ: dict, start_response: StartResponse) -> Iterable[bytes]:
    user = environ["treasury.user"]
    document_id = environ["treasury.document_id"]
    with connect() as conn:
        row = conn.execute(
            """
            SELECT filename, content_type, storage_path
              FROM documents
             WHERE id = ? AND organization_id = ?
            """,
            (document_id, user["organization_id"]),
        ).fetchone()
    if not row:
        return json_response(start_response, 404, {"error": "document_not_found"})
    storage_path = Path(row["storage_path"])
    if not storage_path.exists():
        return json_response(start_response, 410, {"error": "document_missing_from_storage"})
    headers = [
        ("Content-Type", row["content_type"]),
        ("Content-Disposition", f'attachment; filename="{row["filename"]}"'),
    ]
    return response(start_response, 200, storage_path.read_bytes(), headers)


def index_worker(stop_event: threading.Event) -> None:
    while not stop_event.wait(4):
        try:
            with connect() as conn:
                row = conn.execute(
                    """
                    SELECT id
                      FROM documents
                     WHERE status = 'uploaded'
                     ORDER BY created_at
                     LIMIT 1
                    """
                ).fetchone()
                if not row:
                    continue
                conn.execute(
                    "UPDATE documents SET status = 'indexed', indexed_at = CURRENT_TIMESTAMP WHERE id = ?",
                    (row["id"],),
                )
                conn.execute(
                    """
                    INSERT INTO audit_records
                      (id, organization_id, actor_user_id, service, action, method, path, status_code, metadata)
                    SELECT ?, organization_id, NULL, 'docs-worker', 'document.indexed', 'WORKER', ?, 200, ?
                      FROM documents
                     WHERE id = ?
                    """,
                    (
                        f"audit-{uuid.uuid4()}",
                        f"/documents/{row['id']}",
                        json.dumps({"document_id": row["id"]}),
                        row["id"],
                    ),
                )
        except Exception as exc:
            print(f"docs-worker indexer failed: {exc}")


def action_for(environ: dict) -> str:
    method = environ.get("REQUEST_METHOD", "")
    path = environ.get("PATH_INFO", "")
    if method == "POST" and path == "/documents":
        return "document.upload"
    if method == "GET" and path.endswith("/download"):
        return "document.download"
    if method == "GET" and path == "/documents":
        return "document.list"
    return "request"


def safe_filename(value: str) -> str:
    base = Path(value).name
    return re.sub(r"[^A-Za-z0-9._-]", "_", base)[:120] or "treasury-document.bin"


def json_response(start_response: StartResponse, status: int, payload: dict) -> Iterable[bytes]:
    return response(
        start_response,
        status,
        json.dumps(payload).encode(),
        [("Content-Type", "application/json")],
    )


def response(start_response: StartResponse, status: int, body: bytes, headers: list[tuple[str, str]]) -> Iterable[bytes]:
    status_text = {
        200: "OK",
        201: "Created",
        204: "No Content",
        400: "Bad Request",
        401: "Unauthorized",
        403: "Forbidden",
        404: "Not Found",
        410: "Gone",
        500: "Internal Server Error",
    }.get(status, "OK")
    start_response(f"{status} {status_text}", [("Content-Length", str(len(body))), *headers])
    return [body]


def build_app() -> WSGIApp:
    return RequestContextMiddleware(CorsMiddleware(AuditMiddleware(AuthMiddleware(Router()))))


if __name__ == "__main__":
    init_db()
    stop = threading.Event()
    thread = threading.Thread(target=index_worker, args=(stop,), daemon=True)
    thread.start()
    try:
        with make_server("", PORT, build_app()) as server:
            print(f"docs worker listening on http://localhost:{PORT}")
            server.serve_forever()
    finally:
        stop.set()

class ResolveWithinTests(unittest.TestCase):
    def setUp(self):
        self._tmp = tempfile.TemporaryDirectory()
        self.base = Path(self._tmp.name) / "base"
        self.base.mkdir()
        self.outside = Path(self._tmp.name) / "outside.txt"
        self.outside.write_text("outside")

    def test_legitimate_file_inside_root(self):

        target.write_text("CREATE TABLE t (id INTEGER);")

        link = self.base / "schema.sql"
        os.symlink(self.outside, link)

func (w *auditWriter) WriteHeader(status int) {
	w.status = status
	w.ResponseWriter.WriteHeader(status)

func main() {
	root, err := findRoot()
	if err != nil {
		log.Fatal(err)
	}
	db, err := openDB(root)
	if err != nil {
		log.Fatal(err)
	}
	defer db.Close()

	a := &app{db: db, startedAt: time.Now()}
	go a.settlementWorker(context.Background())

	mux := http.NewServeMux()
	mux.HandleFunc("GET /health", a.health)
	mux.HandleFunc("GET /metrics", a.metrics)
	mux.Handle("GET /accounts", a.withMiddleware(http.HandlerFunc(a.accounts), "viewer"))
	mux.Handle("GET /beneficiaries", a.withMiddleware(http.HandlerFunc(a.beneficiaries), "viewer"))
	mux.Handle("GET /transfers", a.withMiddleware(http.HandlerFunc(a.transfers), "viewer"))
	mux.Handle("POST /transfers", a.withMiddleware(http.HandlerFunc(a.createTransfer), "operator"))
	mux.Handle("GET /transfers/{id}/receipt", a.withMiddleware(http.HandlerFunc(a.transferReceipt), "viewer"))

	port := getenv("PORT", "8080")
	log.Printf("payments api listening on http://localhost:%s", port)
	log.Fatal(http.ListenAndServe(":"+port, mux))
}

func findRoot() (string, error) {
	wd, err := os.Getwd()
	if err != nil {
		return "", err
	}
	for {
		if _, err := os.Stat(filepath.Join(wd, "scripts", "schema.sql")); err == nil {
			return wd, nil
		}
		parent := filepath.Dir(wd)
		if parent == wd {
			return "", errors.New("could not find repository root with scripts/schema.sql")
		}
		wd = parent
	}
}

func openDB(root string) (*sql.DB, error) {
	dbPath := getenv("TREASURY_DB_PATH", filepath.Join(root, "data", "treasury.db"))
	if err := os.MkdirAll(filepath.Dir(dbPath), 0o755); err != nil {
		return nil, err
	}
	db, err := sql.Open("sqlite", dbPath)
	if err != nil {
		return nil, err
	}
	db.SetMaxOpenConns(1)
	schema, err := os.ReadFile(filepath.Join(root, "scripts", "schema.sql"))
	if err != nil {
		return nil, err
	}
	if _, err := db.Exec(string(schema)); err != nil {
		return nil, err
	}
	return db, nil
}

func (a *app) withMiddleware(next http.Handler, minRole string) http.Handler {
	return requestID(cors(a.authenticate(a.audit(a.authorize(next, minRole)))))
}

func requestID(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("X-Request-Id", fmt.Sprintf("req-%d", time.Now().UnixNano()))
		next.ServeHTTP(w, r)
	})
}

func cors(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Access-Control-Allow-Origin", "*")
		w.Header().Set("Access-Control-Allow-Headers", "Content-Type, X-User-Id")
		w.Header().Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
		if r.Method == http.MethodOptions {
			w.WriteHeader(http.StatusNoContent)
			return
		}
		next.ServeHTTP(w, r)
	})
}

func (a *app) authenticate(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		userID := r.Header.Get("X-User-Id")
		if userID == "" {
			userID = "u-alice"
		}
		u, err := a.lookupUser(r.Context(), userID)
		if err != nil {
			writeJSON(w, http.StatusUnauthorized, map[string]string{"error": "invalid_user"})
			return
		}
		next.ServeHTTP(w, r.WithContext(context.WithValue(r.Context(), userKey, u)))
	})
}

func (a *app) authorize(next http.Handler, minRole string) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		u := currentUser(r)
		if roleRank(u.Role) < roleRank(minRole) {
			writeJSON(w, http.StatusForbidden, map[string]string{"error": "forbidden", "requiredRole": minRole})
			return
		}
		next.ServeHTTP(w, r)
	})
}

func (a *app) audit(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		aw := &auditWriter{ResponseWriter: w, status: http.StatusOK}
		next.ServeHTTP(aw, r)
		u := currentUser(r)
		_, err := a.db.ExecContext(
			r.Context(),
			`INSERT INTO audit_records
			 (id, organization_id, actor_user_id, service, action, method, path, status_code, metadata)
			 VALUES (?, ?, ?, 'payments-api', ?, ?, ?, ?, ?)`,
			newID("audit"),
			u.OrganizationID,
			u.ID,
			actionFor(r),
			r.Method,
			r.URL.Path,
			aw.status,
			fmt.Sprintf(`{"remote_addr":%q}`, r.RemoteAddr),
		)
		if err != nil {
			log.Printf("audit insert failed: %v", err)
		}
	})
}

func (a *app) lookupUser(ctx context.Context, id string) (user, error) {
	var u user
	err := a.db.QueryRowContext(ctx,
		`SELECT id, organization_id, name, email, role
		   FROM users
		  WHERE id = ? AND active = 1`,
		id,
	).Scan(&u.ID, &u.OrganizationID, &u.Name, &u.Email, &u.Role)
	return u, err
}

func currentUser(r *http.Request) user {
	u, _ := r.Context().Value(userKey).(user)
	return u
}

func roleRank(role string) int {
	switch role {
	case "admin":
		return 4
	case "treasury_manager":
		return 3
	case "operator":
		return 2
	case "viewer":
		return 1
	default:
		return 0
	}
}

func actionFor(r *http.Request) string {
	switch {
	case r.Method == http.MethodPost && r.URL.Path == "/transfers":
		return "transfer.create"
	case strings.HasSuffix(r.URL.Path, "/receipt"):
		return "transfer.receipt.download"
	default:
		return "request"
	}
}

func (a *app) createTransfer(w http.ResponseWriter, r *http.Request) {
	var input transferRequest
	if err := json.NewDecoder(r.Body).Decode(&input); err != nil {
		writeJSON(w, http.StatusBadRequest, map[string]string{"error": "invalid_json"})
		return
	}
	if input.SourceAccountID == "" || input.BeneficiaryID == "" || input.AmountCents <= 0 || input.Currency == "" {
		writeJSON(w, http.StatusBadRequest, map[string]string{"error": "missing_required_fields"})
		return
	}

	u := currentUser(r)
	tx, err := a.db.BeginTx(r.Context(), nil)
	if err != nil {
		writeJSON(w, http.StatusInternalServerError, map[string]string{"error": err.Error()})
		return
	}
	defer tx.Rollback()

	var balance int64
	var accountCurrency string
	err = tx.QueryRowContext(r.Context(),
		`SELECT balance_cents, currency
		   FROM accounts
		  WHERE id = ? AND organization_id = ? AND status = 'open'`,
		input.SourceAccountID,
		u.OrganizationID,
	).Scan(&balance, &accountCurrency)
	if err != nil {
		writeJSON(w, http.StatusBadRequest, map[string]string{"error": "source_account_not_found"})
		return
	}
	if accountCurrency != input.Currency {
		writeJSON(w, http.StatusBadRequest, map[string]string{"error": "currency_mismatch"})
		return
	}
	if balance < input.AmountCents {
		writeJSON(w, http.StatusBadRequest, map[string]string{"error": "insufficient_funds"})
		return
	}

	var beneficiaryCount int
	err = tx.QueryRowContext(r.Context(),
		`SELECT COUNT(*)
		   FROM beneficiaries
		  WHERE id = ? AND organization_id = ? AND currency = ? AND status = 'active'`,
		input.BeneficiaryID,
		u.OrganizationID,
		input.Currency,
	).Scan(&beneficiaryCount)
	if err != nil || beneficiaryCount == 0 {
		writeJSON(w, http.StatusBadRequest, map[string]string{"error": "beneficiary_not_found"})
		return
	}

	transferID := newID("tr")
	jobID := newID("job")
	_, err = tx.ExecContext(r.Context(),
		`INSERT INTO transfers
		 (id, organization_id, source_account_id, beneficiary_id, amount_cents, currency, memo, status, requested_by)
		 VALUES (?, ?, ?, ?, ?, ?, ?, 'pending', ?)`,
		transferID,
		u.OrganizationID,
		input.SourceAccountID,
		input.BeneficiaryID,
		input.AmountCents,
		input.Currency,
		input.Memo,
		u.ID,
	)
	if err != nil {
		writeJSON(w, http.StatusInternalServerError, map[string]string{"error": err.Error()})
		return
	}
	_, err = tx.ExecContext(r.Context(),
		`INSERT INTO settlement_jobs (id, organization_id, transfer_id, status)
		 VALUES (?, ?, ?, 'queued')`,
		jobID,
		u.OrganizationID,
		transferID,
	)
	if err != nil {
		writeJSON(w, http.StatusInternalServerError, map[string]string{"error": err.Error()})
		return
	}
	if err := tx.Commit(); err != nil {
		writeJSON(w, http.StatusInternalServerError, map[string]string{"error": err.Error()})
		return
	}
	writeJSON(w, http.StatusAccepted, map[string]any{"id": transferID, "status": "pending", "settlement_job_id": jobID})
}

		createdAt,
		nullText(settledAt),
	)
}

func (a *app) settlementWorker(ctx context.Context) {
	ticker := time.NewTicker(3 * time.Second)
	defer ticker.Stop()
	for {
		select {
		case <-ctx.Done():
			return
		case <-ticker.C:
			if err := a.processSettlementJob(ctx); err != nil && !errors.Is(err, sql.ErrNoRows) {
				log.Printf("settlement worker error: %v", err)
			}
		}
	}
}

func (a *app) processSettlementJob(ctx context.Context) error {
	tx, err := a.db.BeginTx(ctx, nil)
	if err != nil {
		return err
	}
	defer tx.Rollback()

	var jobID, orgID, transferID, sourceAccountID string
	var amount int64
	err = tx.QueryRowContext(ctx,
		`SELECT sj.id, sj.organization_id, sj.transfer_id, tr.source_account_id, tr.amount_cents
		   FROM settlement_jobs sj
		   JOIN transfers tr ON tr.id = sj.transfer_id
		  WHERE sj.status = 'queued'
		  ORDER BY sj.created_at
		  LIMIT 1`,
	).Scan(&jobID, &orgID, &transferID, &sourceAccountID, &amount)
	if err != nil {
		return err
	}

	now := time.Now().UTC().Format(time.RFC3339)
	_, err = tx.ExecContext(ctx,
		`UPDATE settlement_jobs
		    SET status = 'processing', attempts = attempts + 1, updated_at = ?
		  WHERE id = ? AND status = 'queued'`,
		now,
		jobID,
	)
	if err != nil {
		return err
	}
	_, err = tx.ExecContext(ctx,
		`UPDATE accounts
		    SET balance_cents = balance_cents - ?
		  WHERE id = ? AND organization_id = ? AND balance_cents >= ?`,
		amount,
		sourceAccountID,
		orgID,
		amount,
	)
	if err != nil {
		return err
	}
	_, err = tx.ExecContext(ctx,
		`UPDATE transfers SET status = 'settled', settled_at = ? WHERE id = ? AND status = 'pending'`,
		now,
		transferID,
	)
	if err != nil {
		return err
	}
	_, err = tx.ExecContext(ctx,
		`UPDATE settlement_jobs SET status = 'done', updated_at = ? WHERE id = ?`,
		now,
		jobID,
	)
	if err != nil {
		return err
	}
	_, err = tx.ExecContext(ctx,
		`INSERT INTO audit_records
		 (id, organization_id, actor_user_id, service, action, method, path, status_code, metadata)
		 VALUES (?, ?, NULL, 'payments-api', 'settlement.complete', 'WORKER', ?, 200, ?)`,
		newID("audit"),
		orgID,
		"/settlement-jobs/"+jobID,
		fmt.Sprintf(`{"transfer_id":%q}`, transferID),
	)
	if err != nil {
		return err
	}
	return tx.Commit()
}

func rowsToMaps(rows *sql.Rows) ([]map[string]any, error) {
	cols, err := rows.Columns()
	if err != nil {
		return nil, err
	}
	var result []map[string]any
	for rows.Next() {
		values := make([]any, len(cols))
		ptrs := make([]any, len(cols))
		for i := range values {
			ptrs[i] = &values[i]
		}
		if err := rows.Scan(ptrs...); err != nil {
			return nil, err
		}
		item := map[string]any{}
		for i, col := range cols {
			switch v := values[i].(type) {
			case []byte:
				item[col] = string(v)
			default:
				item[col] = v
			}
		}
		result = append(result, item)
	}
	return result, rows.Err()
}

func getenv(key, fallback string) string {

function table(headers, rows) {
  if (!rows.length) {
    return `<div class="list"><div class="subtle">No records</div></div>`;
  }
  return `<table><thead><tr>${headers.map((header) => `<th>${escapeHtml(header)}</th>`).join("")}</tr></thead><tbody>${rows
    .map((row) => `<tr>${row.map((cell) => `<td>${renderCell(cell)}</td>`).join("")}</tr>`)
    .join("")}</tbody></table>`;
}

function renderCell(cell) {
  if (cell !== null && typeof cell === "object" && typeof cell.html === "string") {
    return cell.html;
  }
  return escapeHtml(cell);
}

function escapeHtml(value) {
  return String(value)
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}

import type { NextFunction, Request, Response } from "express";
import { randomUUID } from "node:crypto";
import { audit, AuthUser, getUser } from "./db";

declare global {
  namespace Express {
    interface Request {
      requestId: string;
      user?: AuthUser;
    }
  }
}

const roleRank = {
  viewer: 1,
  operator: 2,
  treasury_manager: 3,
  admin: 4
};

export function requestContext(req: Request, res: Response, next: NextFunction) {
  req.requestId = randomUUID();
  res.setHeader("X-Request-Id", req.requestId);
  next();
}

export function authenticate(req: Request, res: Response, next: NextFunction) {
  const userId = String(req.header("X-User-Id") ?? "u-alice");
  const user = getUser(userId);
  if (!user) {
    return res.status(401).json({ error: "invalid_user" });
  }
  req.user = user;
  next();
}

export function requireRole(minRole: keyof typeof roleRank) {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!req.user || roleRank[req.user.role] < roleRank[minRole]) {
      return res.status(403).json({ error: "forbidden", requiredRole: minRole });
    }
    next();
  };
}

export function auditLogger(req: Request, res: Response, next: NextFunction) {
  const started = Date.now();
  res.on("finish", () => {
    if (req.path === "/favicon.ico") {
      return;
    }
    try {
      audit({
        id: randomUUID(),
        organizationId: req.user?.organization_id,
        actorUserId: req.user?.id,
        action: routeAction(req),
        method: req.method,
        path: req.originalUrl,
        statusCode: res.statusCode,
        metadata: {
          requestId: req.requestId,
          durationMs: Date.now() - started,
          userAgent: req.header("user-agent")
        }
      });
    } catch (error) {
      console.error("portal audit failed", error);
    }
  });
  next();
}

function routeAction(req: Request) {
  if (req.method === "POST" && req.path.includes("beneficiaries")) {
    return "beneficiary.create";
  }
  if (req.path.includes("dashboard")) {
    return "dashboard.read";
  }
  return "request";
}

Java HTTP service for the fictional treasury operations portal.

## Endpoints

- `GET /health`
- `GET /metrics`
- `GET /risk/exposure`
- `GET /risk/limits`
- `POST /risk/audit`

Authenticated endpoints use `X-User-Id` and the shared SQLite user table. If
the header is omitted, the service uses seeded admin user `u-alice`.

## Run

```sh
javac -d build/classes src/com/sthenos/treasury/RiskApi.java
java -cp build/classes com.sthenos.treasury.RiskApi
```

Environment variables:

- `PORT`: defaults to `8070`
- `TREASURY_DB_PATH`: defaults to `../data/treasury.db`

package com.sthenos.treasury;

import com.sun.net.httpserver.Headers;
import com.sun.net.httpserver.HttpExchange;
import com.sun.net.httpserver.HttpHandler;
import com.sun.net.httpserver.HttpServer;

import java.io.IOException;
import java.io.OutputStream;
import java.net.InetSocketAddress;
import java.net.URI;
import java.nio.charset.StandardCharsets;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.time.Instant;
import java.util.ArrayList;
import java.util.List;
import java.util.Locale;
import java.util.UUID;

public final class RiskApi {
    private static final String DEFAULT_DB = "../data/treasury.db";
    private static final int DEFAULT_PORT = 8070;
    private final String dbUrl;
    private final long startedAt = System.currentTimeMillis();

    public RiskApi(String dbPath) {
        this.dbUrl = "jdbc:sqlite:" + dbPath;
    }

    public static void main(String[] args) throws Exception {
        String dbPath = System.getenv().getOrDefault("TREASURY_DB_PATH", DEFAULT_DB);
        int port = Integer.parseInt(System.getenv().getOrDefault("PORT", String.valueOf(DEFAULT_PORT)));
        RiskApi app = new RiskApi(dbPath);
        HttpServer server = HttpServer.create(new InetSocketAddress(port), 0);
        server.createContext("/health", app::health);
        server.createContext("/metrics", app::metrics);
        server.createContext("/risk/exposure", app.withAuth("viewer", app::riskExposure));
        server.createContext("/risk/limits", app.withAuth("treasury_manager", app::riskLimits));
        server.createContext("/risk/audit", app.withAuth("admin", app::riskAudit));
        server.start();
        System.out.println("treasury risk-api listening on http://localhost:" + port);
    }

    private HttpHandler withAuth(String minRole, AuthedHandler handler) {
        return exchange -> {
            User user = authenticate(exchange);
            if (user == null) {
                respond(exchange, 401, "{\"error\":\"invalid_user\"}");
                return;
            }
            if (!hasRole(user.role(), minRole)) {
                audit(exchange, user, "risk-api", "access.denied", 403, "{\"required\":\"" + minRole + "\"}");
                respond(exchange, 403, "{\"error\":\"forbidden\"}");
                return;
            }
            try {
                handler.handle(exchange, user);
            } finally {
                audit(exchange, user, "risk-api", routeAction(exchange), 200, "{}");
            }
        };
    }

    private void health(HttpExchange exchange) throws IOException {
        respond(exchange, 200, "{\"ok\":true,\"service\":\"risk-api\",\"uptimeSeconds\":" + uptimeSeconds() + "}");
    }

    private void metrics(HttpExchange exchange) throws IOException {
        String body;
        try (Connection conn = connect()) {
            int openLimits = count(conn, "SELECT COUNT(*) FROM accounts WHERE status = 'open'");
            int pendingTransfers = count(conn, "SELECT COUNT(*) FROM transfers WHERE status = 'pending'");
            body = "# HELP treasury_risk_open_accounts Open accounts\n"
                + "# TYPE treasury_risk_open_accounts gauge\n"
                + "treasury_risk_open_accounts " + openLimits + "\n"
                + "# HELP treasury_risk_pending_transfers Pending transfers\n"
                + "# TYPE treasury_risk_pending_transfers gauge\n"
                + "treasury_risk_pending_transfers " + pendingTransfers + "\n";
        } catch (SQLException exc) {
            respond(exchange, 500, "{\"error\":\"db_error\"}");
            return;
        }
        Headers headers = exchange.getResponseHeaders();
        headers.add("Content-Type", "text/plain; charset=utf-8");
        write(exchange, 200, body);
    }

    private void riskExposure(HttpExchange exchange, User user) throws IOException {
        if (!"GET".equals(exchange.getRequestMethod())) {
            respond(exchange, 405, "{\"error\":\"method_not_allowed\"}");
            return;
        }
        try (Connection conn = connect()) {
            String exposure = queryExposure(conn, user.organizationId());
            respond(exchange, 200, exposure);
        } catch (SQLException exc) {
            respond(exchange, 500, "{\"error\":\"db_error\"}");
        }
    }

    private void riskLimits(HttpExchange exchange, User user) throws IOException {
        if (!"GET".equals(exchange.getRequestMethod())) {
            respond(exchange, 405, "{\"error\":\"method_not_allowed\"}");
            return;
        }
        try (Connection conn = connect()) {
            String limits = queryLimits(conn, user.organizationId());
            respond(exchange, 200, limits);
        } catch (SQLException exc) {
            respond(exchange, 500, "{\"error\":\"db_error\"}");
        }
    }

    private void riskAudit(HttpExchange exchange, User user) throws IOException {
        if (!"POST".equals(exchange.getRequestMethod())) {
            respond(exchange, 405, "{\"error\":\"method_not_allowed\"}");
            return;
        }
        audit(exchange, user, "risk-api", "risk.manual_review", 202, "{\"source\":\"manual\"}");
        respond(exchange, 202, "{\"status\":\"queued\",\"organization_id\":\"" + escape(user.organizationId()) + "\"}");
    }

    private User authenticate(HttpExchange exchange) {
        String userId = exchange.getRequestHeaders().getFirst("X-User-Id");
        if (userId == null || userId.isBlank()) {
            userId = "u-alice";
        }
        try (Connection conn = connect();
             PreparedStatement stmt = conn.prepareStatement(
                 "SELECT id, organization_id, team_id, email, name, role FROM users WHERE id = ?")) {
            stmt.setString(1, userId);
            try (ResultSet rows = stmt.executeQuery()) {
                if (!rows.next()) {
                    return null;
                }
                return new User(
                    rows.getString("id"),
                    rows.getString("organization_id"),
                    rows.getString("team_id"),
                    rows.getString("email"),
                    rows.getString("name"),
                    rows.getString("role")
                );
            }
        } catch (SQLException exc) {
            return null;
        }
    }

    private String queryExposure(Connection conn, String orgId) throws SQLException {
        List<String> rows = new ArrayList<>();
        try (PreparedStatement stmt = conn.prepareStatement(
            "SELECT a.id, a.name, a.balance_cents, "
                + "COALESCE(SUM(CASE WHEN t.status = 'pending' THEN t.amount_cents ELSE 0 END), 0) AS pending_cents "
                + "FROM accounts a LEFT JOIN transfers t ON t.source_account_id = a.id AND t.organization_id = a.organization_id "
                + "WHERE a.organization_id = ? GROUP BY a.id, a.name, a.balance_cents ORDER BY a.name")) {
            stmt.setString(1, orgId);
            try (ResultSet rs = stmt.executeQuery()) {
                while (rs.next()) {
                    rows.add("{\"account_id\":\"" + escape(rs.getString("id")) + "\","
                        + "\"name\":\"" + escape(rs.getString("name")) + "\","
                        + "\"balance_cents\":" + rs.getLong("balance_cents") + ","
                        + "\"pending_transfer_cents\":" + rs.getLong("pending_cents") + "}");
                }
            }
        }
        return "{\"organization_id\":\"" + escape(orgId) + "\",\"exposures\":[" + String.join(",", rows) + "]}";
    }

    private String queryLimits(Connection conn, String orgId) throws SQLException {
        long transferCount = 0;
        long pendingTotal = 0;
        try (PreparedStatement stmt = conn.prepareStatement(
            "SELECT COUNT(*) AS transfer_count, COALESCE(SUM(amount_cents), 0) AS pending_total "
                + "FROM transfers WHERE organization_id = ? AND status = 'pending'")) {
            stmt.setString(1, orgId);
            try (ResultSet rs = stmt.executeQuery()) {
                if (rs.next()) {
                    transferCount = rs.getLong("transfer_count");
                    pendingTotal = rs.getLong("pending_total");
                }
            }
        }
        String state = pendingTotal > 5000000 ? "review_required" : "within_limit";
        return "{\"organization_id\":\"" + escape(orgId) + "\",\"pending_transfer_count\":" + transferCount
            + ",\"pending_transfer_cents\":" + pendingTotal + ",\"state\":\"" + state + "\"}";
    }

    private void audit(HttpExchange exchange, User user, String service, String action, int status, String metadata) {
        try (Connection conn = connect();
             PreparedStatement stmt = conn.prepareStatement(
                 "INSERT INTO audit_records "
                     + "(id, organization_id, actor_user_id, service, action, method, path, status_code, metadata) "
                     + "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)")) {
            URI uri = exchange.getRequestURI();
            stmt.setString(1, "audit-" + UUID.randomUUID());
            stmt.setString(2, user.organizationId());
            stmt.setString(3, user.id());
            stmt.setString(4, service);
            stmt.setString(5, action);
            stmt.setString(6, exchange.getRequestMethod());
            stmt.setString(7, uri.getPath());
            stmt.setInt(8, status);
            stmt.setString(9, metadata);
            stmt.executeUpdate();
        } catch (SQLException ignored) {
            // Audit must not break the API response path.
        }
    }

    private Connection connect() throws SQLException {
        return DriverManager.getConnection(dbUrl);
    }

    private int count(Connection conn, String sql) throws SQLException {
        try (PreparedStatement stmt = conn.prepareStatement(sql);
             ResultSet rs = stmt.executeQuery()) {
            return rs.next() ? rs.getInt(1) : 0;
        }
    }

    private String routeAction(HttpExchange exchange) {
        String path = exchange.getRequestURI().getPath();
        if (path.endsWith("/exposure")) {
            return "risk.exposure";
        }
        if (path.endsWith("/limits")) {
            return "risk.limits";
        }
        return "risk.request";
    }

    private boolean hasRole(String actual, String required) {
        return rank(actual) >= rank(required);
    }

    private int rank(String role) {
        return switch (String.valueOf(role).toLowerCase(Locale.ROOT)) {
            case "admin" -> 4;
            case "treasury_manager" -> 3;
            case "operator" -> 2;
            case "viewer" -> 1;
            default -> 0;
        };
    }

    private long uptimeSeconds() {
        return Math.max(0, (System.currentTimeMillis() - startedAt) / 1000);
    }

    private void respond(HttpExchange exchange, int status, String body) throws IOException {
        exchange.getResponseHeaders().add("Content-Type", "application/json; charset=utf-8");
        write(exchange, status, body);
    }

    private void write(HttpExchange exchange, int status, String body) throws IOException {
        byte[] bytes = body.getBytes(StandardCharsets.UTF_8);
        exchange.sendResponseHeaders(status, bytes.length);
        try (OutputStream output = exchange.getResponseBody()) {
            output.write(bytes);
        }
    }

    private String escape(String value) {
        return String.valueOf(value)
            .replace("\\", "\\\\")
            .replace("\"", "\\\"");
    }

    @FunctionalInterface
    private interface AuthedHandler {
        void handle(HttpExchange exchange, User user) throws IOException;
    }

    private record User(String id, String organizationId, String teamId, String email, String name, String role) {}
}

from pathlib import Path
import sqlite3


ROOT = Path(__file__).resolve().parents[1]
DB_PATH = ROOT / "data" / "treasury.db"
SCHEMA_PATH = ROOT / "scripts" / "schema.sql"


def main() -> None:
    DB_PATH.parent.mkdir(parents=True, exist_ok=True)
    with sqlite3.connect(DB_PATH) as conn:
        conn.executescript(SCHEMA_PATH.read_text())
    print(f"initialized {DB_PATH}")


if __name__ == "__main__":
    main()

PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS organizations (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  plan TEXT NOT NULL,
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS teams (
  id TEXT PRIMARY KEY,
  organization_id TEXT NOT NULL REFERENCES organizations(id),
  name TEXT NOT NULL,
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS users (
  id TEXT PRIMARY KEY,
  organization_id TEXT NOT NULL REFERENCES organizations(id),
  team_id TEXT REFERENCES teams(id),
  email TEXT NOT NULL UNIQUE,
  name TEXT NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('admin', 'treasury_manager', 'operator', 'viewer')),
  active INTEGER NOT NULL DEFAULT 1,
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS accounts (
  id TEXT PRIMARY KEY,
  organization_id TEXT NOT NULL REFERENCES organizations(id),
  name TEXT NOT NULL,
  bank_name TEXT NOT NULL,
  currency TEXT NOT NULL,
  balance_cents INTEGER NOT NULL DEFAULT 0,
  status TEXT NOT NULL DEFAULT 'open',
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS beneficiaries (
  id TEXT PRIMARY KEY,
  organization_id TEXT NOT NULL REFERENCES organizations(id),
  name TEXT NOT NULL,
  bank_name TEXT NOT NULL,
  routing_number TEXT NOT NULL,
  account_number_last4 TEXT NOT NULL,
  currency TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'active',
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS transfers (
  id TEXT PRIMARY KEY,
  organization_id TEXT NOT NULL REFERENCES organizations(id),
  source_account_id TEXT NOT NULL REFERENCES accounts(id),
  beneficiary_id TEXT NOT NULL REFERENCES beneficiaries(id),
  amount_cents INTEGER NOT NULL,
  currency TEXT NOT NULL,
  memo TEXT,
  status TEXT NOT NULL DEFAULT 'pending',
  requested_by TEXT NOT NULL REFERENCES users(id),
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  settled_at TEXT
);

CREATE TABLE IF NOT EXISTS settlement_jobs (
  id TEXT PRIMARY KEY,
  organization_id TEXT NOT NULL REFERENCES organizations(id),
  transfer_id TEXT NOT NULL REFERENCES transfers(id),
  status TEXT NOT NULL DEFAULT 'queued',
  attempts INTEGER NOT NULL DEFAULT 0,
  last_error TEXT,
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS documents (
  id TEXT PRIMARY KEY,
  organization_id TEXT NOT NULL REFERENCES organizations(id),
  uploaded_by TEXT NOT NULL REFERENCES users(id),
  filename TEXT NOT NULL,
  content_type TEXT NOT NULL,
  storage_path TEXT NOT NULL,
  size_bytes INTEGER NOT NULL,
  status TEXT NOT NULL DEFAULT 'uploaded',
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  indexed_at TEXT
);

CREATE TABLE IF NOT EXISTS audit_records (
  id TEXT PRIMARY KEY,
  organization_id TEXT,
  actor_user_id TEXT,
  service TEXT NOT NULL,
  action TEXT NOT NULL,
  method TEXT NOT NULL,
  path TEXT NOT NULL,
  status_code INTEGER NOT NULL,
  metadata TEXT,
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_users_org ON users(organization_id);
CREATE INDEX IF NOT EXISTS idx_accounts_org ON accounts(organization_id);
CREATE INDEX IF NOT EXISTS idx_beneficiaries_org ON beneficiaries(organization_id);
CREATE INDEX IF NOT EXISTS idx_transfers_org_status ON transfers(organization_id, status);
CREATE INDEX IF NOT EXISTS idx_audit_org_created ON audit_records(organization_id, created_at);
CREATE INDEX IF NOT EXISTS idx_documents_org ON documents(organization_id);
CREATE INDEX IF NOT EXISTS idx_settlement_jobs_status ON settlement_jobs(status, updated_at);
