+def resolve_within(base: Path, candidate: Path, label: str) -> Path:
+    """Fully resolve `candidate` and require the result to stay inside `base`.
+
+    `Path.resolve()` collapses `..` segments and follows every symlink in the
+    portion of the path that already exists, so an attacker-planted symlink
+    (e.g. `data/treasury.db` -> `/etc/shadow`, or an absolute/traversal value in
+    `TREASURY_DB_PATH`) resolves to its real target before the containment check
+    runs. The resolved path is returned so callers open the real location rather
+    than the symlink, which also closes the resolve/open TOCTOU window.
+    """
+    base_resolved = base.resolve()
+    resolved = Path(candidate).expanduser().resolve()
+    if resolved != base_resolved and base_resolved not in resolved.parents:
+        raise UnsafeStoragePath(f"{label} resolves outside the trusted data root")
+    return resolved
